cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico CACHE STRING "Target board")
set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/.deps/pico-sdk" CACHE PATH "Pinned Pico SDK")
set(picotool_DIR "${CMAKE_CURRENT_LIST_DIR}/.deps/picotool-install/picotool" CACHE PATH "Pinned host picotool")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(nitro_ps3 C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
option(NITRO_DEBUG "Nonblocking UART1 debug TX on GP4" OFF)
option(NITRO_PRO_DRUMS "Experimental Santroller Pro cymbal encoding" OFF)
add_executable(nitro_ps3 src/main.c src/core.c src/usb_descriptors.c)
target_include_directories(nitro_ps3 PRIVATE include third_party/santroller)
target_compile_definitions(nitro_ps3 PRIVATE PICO_NO_BI_PROGRAM_BUILD_DATE=1 NITRO_DEBUG=$<BOOL:${NITRO_DEBUG}> NITRO_PRO_DRUMS=$<BOOL:${NITRO_PRO_DRUMS}>)
target_compile_options(nitro_ps3 PRIVATE -Wall -Wextra)
target_link_libraries(nitro_ps3 pico_stdlib pico_util hardware_uart hardware_irq hardware_sync tinyusb_device tinyusb_board)
pico_enable_stdio_usb(nitro_ps3 0)
pico_enable_stdio_uart(nitro_ps3 0)
pico_add_extra_outputs(nitro_ps3)
