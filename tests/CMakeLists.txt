cmake_minimum_required(VERSION 3.20)
project(nitro_tests C)
enable_testing()
foreach(mode IN ITEMS standard pro)
 add_executable(test_${mode} test_core.c ../src/core.c)
 target_include_directories(test_${mode} PRIVATE ../include)
 target_compile_options(test_${mode} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 if(mode STREQUAL "pro")
  target_compile_definitions(test_${mode} PRIVATE NITRO_PRO_DRUMS=1)
 endif()
 add_test(NAME ${mode} COMMAND test_${mode})
endforeach()

foreach(kind IN ITEMS local upstream)
 add_executable(dump_${kind} ../src/usb_descriptors.c)
 target_compile_definitions(dump_${kind} PRIVATE DESCRIPTOR_DUMP)
 target_include_directories(dump_${kind} PRIVATE ../include ../third_party/santroller ../.deps/pico-sdk/lib/tinyusb/src ${CMAKE_BINARY_DIR}/generated)
 if(kind STREQUAL "upstream")
  target_compile_definitions(dump_${kind} PRIVATE UPSTREAM_DESCRIPTOR)
 endif()
endforeach()

add_executable(test_protocol test_protocol.c ../src/core.c)
target_include_directories(test_protocol PRIVATE ../include ${CMAKE_BINARY_DIR}/generated)
target_compile_options(test_protocol PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME protocol COMMAND test_protocol)
